-- AI floorplan recognition review runs and immutable approval audit.
create table if not exists public.booth_floorplan_detection_runs (
  id uuid primary key default gen_random_uuid(),
  exhibition_id uuid not null references public.exhibitions(id) on delete cascade,
  hall_id uuid references public.exhibition_halls(id) on delete set null,
  source_file_name text not null,
  source_mime_type text not null,
  status text not null default 'review',
  detection_summary jsonb not null default '{}'::jsonb,
  created_by uuid not null default auth.uid() references auth.users(id),
  created_at timestamptz not null default now(),
  approved_at timestamptz,
  approved_by uuid references auth.users(id),
  constraint booth_floorplan_detection_runs_status_check check (status in ('review','approved','cancelled'))
);
create index if not exists booth_floorplan_detection_runs_event_idx on public.booth_floorplan_detection_runs(exhibition_id, created_at desc);
alter table public.booth_floorplan_detection_runs enable row level security;
drop policy if exists booth_floorplan_detection_runs_staff_all on public.booth_floorplan_detection_runs;
create policy booth_floorplan_detection_runs_staff_all on public.booth_floorplan_detection_runs
for all to authenticated
using (private.is_internal_user())
with check (private.is_internal_user());

grant select, insert, update, delete on public.booth_floorplan_detection_runs to authenticated;

-- Apply a reviewed recognition batch atomically: either all booth updates and the
-- audit approval commit, or none of them do.
create or replace function public.approve_booth_floorplan_detection(
  p_run_id uuid,
  p_detections jsonb
) returns void
language plpgsql
security invoker
set search_path = public, private
as $$
declare
  v_run public.booth_floorplan_detection_runs%rowtype;
  v_item jsonb;
  v_booth_id uuid;
  v_number text;
begin
  if not private.is_internal_user() then
    raise exception 'Only VEGA internal users can approve floorplan detections';
  end if;
  select * into v_run from public.booth_floorplan_detection_runs where id=p_run_id for update;
  if not found or v_run.status <> 'review' then
    raise exception 'Recognition run is missing or no longer awaiting review';
  end if;
  if jsonb_typeof(p_detections) <> 'array' or jsonb_array_length(p_detections)=0 then
    raise exception 'At least one reviewed booth is required';
  end if;
  if exists (
    select 1 from jsonb_array_elements(p_detections) d
    group by upper(regexp_replace(trim(d->>'booth_number'),'[^A-Z0-9]','','g'))
    having count(*) > 1
  ) then raise exception 'Duplicate booth numbers in detection batch'; end if;
  for v_item in select value from jsonb_array_elements(p_detections)
  loop
    v_number := trim(v_item->>'booth_number');
    if v_number is null or v_number='' then raise exception 'Booth number cannot be empty'; end if;
    v_booth_id := nullif(v_item->>'booth_id','')::uuid;
    if v_booth_id is not null then
      if exists(select 1 from public.booths where exhibition_id=v_run.exhibition_id and id<>v_booth_id and upper(regexp_replace(booth_number,'[^A-Z0-9]','','g'))=upper(regexp_replace(v_number,'[^A-Z0-9]','','g'))) then
        raise exception 'Booth number % already exists in this exhibition',v_number;
      end if;
      update public.booths set
        booth_number=v_number, hall_id=coalesce(v_run.hall_id,hall_id),
        x_position=(v_item->>'left')::numeric, y_position=(v_item->>'top')::numeric,
        width_percent=(v_item->>'width')::numeric, height_percent=(v_item->>'height')::numeric,
        geometry_type='rectangle', geometry_points=v_item->'geometry_points',
        detection_source='ocr_floorplan', detection_confidence=(v_item->>'confidence')::numeric,
        detection_review_status='approved', updated_at=now()
      where id=v_booth_id and exhibition_id=v_run.exhibition_id;
      if not found then raise exception 'Matched booth not found in selected exhibition'; end if;
    else
      if exists(select 1 from public.booths where exhibition_id=v_run.exhibition_id and upper(regexp_replace(booth_number,'[^A-Z0-9]','','g'))=upper(regexp_replace(v_number,'[^A-Z0-9]','','g'))) then
        raise exception 'Booth number % already exists in this exhibition',v_number;
      end if;
      insert into public.booths(
        exhibition_id,hall_id,booth_number,area_sqm,width_meters,depth_meters,booth_type,status,price,currency,
        x_position,y_position,width_percent,height_percent,geometry_type,geometry_points,detection_source,detection_confidence,detection_review_status
      ) values (
        v_run.exhibition_id,v_run.hall_id,v_number,0,0,0,'Space Only','Available',0,'USD',
        (v_item->>'left')::numeric,(v_item->>'top')::numeric,(v_item->>'width')::numeric,(v_item->>'height')::numeric,
        'rectangle',v_item->'geometry_points','ocr_floorplan',(v_item->>'confidence')::numeric,'approved'
      );
    end if;
  end loop;
  update public.booth_floorplan_detection_runs set status='approved',approved_at=now(),approved_by=auth.uid(),
    detection_summary=detection_summary || jsonb_build_object('approved_count',jsonb_array_length(p_detections))
  where id=p_run_id;
end;
$$;
grant execute on function public.approve_booth_floorplan_detection(uuid,jsonb) to authenticated;
